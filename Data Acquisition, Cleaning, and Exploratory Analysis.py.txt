"""
eda_cleaning.py
----------------
Part 1 — Data Acquisition, Cleaning, and Exploratory Analysis
Dataset: synthetic "Employee HR Analytics" data (data/raw_data.csv)

Run:  python3 eda_cleaning.py
Outputs:
  - cleaned_data.csv
  - plots/*.png  (5 required viz + heatmap)
  - console output used to populate README.md findings
"""

import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import seaborn as sns

plt.rcParams["figure.dpi"] = 110
sns.set_theme(style="whitegrid")

PLOTS = "plots"

def section(title):
    print("\n" + "=" * 80)
    print(title)
    print("=" * 80)

# ---------------------------------------------------------------------------
# Task 1: Load & inspect
# ---------------------------------------------------------------------------
section("TASK 1 — Load & inspect")
df = pd.read_csv("data/raw_data.csv")
print("First 5 rows:\n", df.head())
print("\nDtypes:\n", df.dtypes)
print("\nShape:", df.shape)

# ---------------------------------------------------------------------------
# Task 2: Null value analysis
# ---------------------------------------------------------------------------
section("TASK 2 — Null value analysis")
null_counts = df.isnull().sum()
null_pct = (df.isnull().sum() / df.shape[0]) * 100
null_table = pd.concat([null_counts, null_pct.round(2)], axis=1, keys=["missing_count", "missing_pct"])
print(null_table)

high_null_cols = null_pct[null_pct > 20].index.tolist()
print("\nColumns exceeding 20% nulls:", high_null_cols)

numeric_cols_initial = df.select_dtypes(include=np.number).columns.tolist()
for col in numeric_cols_initial:
    pct = null_pct[col]
    if pct <= 20 and pct > 0:
        med = df[col].median()
        df[col] = df[col].fillna(med)
        print(f"Filled '{col}' nulls with median = {med:.2f}")

print("\nNulls remaining after median-fill of <=20% numeric columns:\n", df.isnull().sum())

# ---------------------------------------------------------------------------
# Task 3: Duplicate detection and removal
# ---------------------------------------------------------------------------
section("TASK 3 — Duplicates")
dup_count = df.duplicated().sum()
print("Duplicate rows found:", dup_count)

null_pct_before_dedup = (df.isnull().sum() / df.shape[0]) * 100
df = df.drop_duplicates().reset_index(drop=True)
null_pct_after_dedup = (df.isnull().sum() / df.shape[0]) * 100
print(f"Rows removed: {dup_count}. New shape: {df.shape}")
print("\nNull % before dedup vs after dedup:")
print(pd.concat([null_pct_before_dedup.round(2), null_pct_after_dedup.round(2)],
                 axis=1, keys=["before_dedup_%", "after_dedup_%"]))

# ---------------------------------------------------------------------------
# Task 4: Data type correction
# ---------------------------------------------------------------------------
section("TASK 4 — Data type correction")
mem_before = df.memory_usage(deep=True).sum()
print(f"Memory usage BEFORE conversion: {mem_before} bytes ({mem_before/1024:.1f} KB)")

# satisfaction_score is object (numeric strings + 'N/A'/'unknown'/'-'/'pending')
df["satisfaction_score"] = pd.to_numeric(df["satisfaction_score"], errors="coerce")
print("\nsatisfaction_score converted to numeric via pd.to_numeric(errors='coerce').")
print("New nulls introduced by coercion (previously non-numeric strings):",
      df["satisfaction_score"].isnull().sum())

print(f"satisfaction_score total null %: {(df['satisfaction_score'].isnull().sum()/df.shape[0]*100):.2f}% "
      f"(>20% -- left unfilled here per Task 2 rule; will be revisited in Task 8/9)")

# department -> category dtype (repetitive string column)
df["department"] = df["department"].astype("category")
df["attrition"] = df["attrition"].astype("category")

mem_after = df.memory_usage(deep=True).sum()
print(f"\nMemory usage AFTER conversion: {mem_after} bytes ({mem_after/1024:.1f} KB)")
print(f"Memory reduction: {mem_before - mem_after} bytes "
      f"({(1 - mem_after/mem_before)*100:.1f}% smaller)")

# ---------------------------------------------------------------------------
# Task 5: Descriptive statistics and skewness
# ---------------------------------------------------------------------------
section("TASK 5 — Descriptive stats & skewness")
numeric_cols = df.select_dtypes(include=np.number).columns.tolist()
numeric_cols = [c for c in numeric_cols if c != "employee_id"]
print(df[numeric_cols].describe())

skew_vals = df[numeric_cols].skew().sort_values(key=lambda s: s.abs(), ascending=False)
print("\nSkewness (sorted by |skew|):\n", skew_vals)
most_skewed_col = skew_vals.index[0]
print(f"\nMost skewed column: {most_skewed_col} (skew = {skew_vals.iloc[0]:.3f})")

# ---------------------------------------------------------------------------
# Task 6: Outlier detection with IQR
# ---------------------------------------------------------------------------
section("TASK 6 — IQR outlier detection")
iqr_cols = ["monthly_salary", "overtime_hours"]
iqr_report = {}
for col in iqr_cols:
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1
    lower = Q1 - 1.5 * IQR
    upper = Q3 + 1.5 * IQR
    n_outliers = ((df[col] < lower) | (df[col] > upper)).sum()
    iqr_report[col] = dict(Q1=Q1, Q3=Q3, IQR=IQR, lower=lower, upper=upper, n_outliers=n_outliers)
    print(f"\n{col}: Q1={Q1:.2f}, Q3={Q3:.2f}, IQR={IQR:.2f}, "
          f"bounds=({lower:.2f}, {upper:.2f}), outliers={n_outliers}")

# ---------------------------------------------------------------------------
# Task 7: Visualizations (5 required types)
# ---------------------------------------------------------------------------
section("TASK 7 — Visualizations")

# 7a. Line plot
plt.figure(figsize=(9, 4.5))
plt.plot(df.sort_values("join_date")["monthly_salary"].reset_index(drop=True))
plt.title("Monthly Salary by Row Order (sorted by join_date)")
plt.xlabel("Row index (sorted by join_date)")
plt.ylabel("Monthly Salary ($)")
plt.tight_layout()
plt.savefig(f"{PLOTS}/01_line_salary.png")
plt.close()

# 7b. Bar chart: mean salary by department
plt.figure(figsize=(8, 5))
dept_mean_salary = df.groupby("department", observed=True)["monthly_salary"].mean().sort_values(ascending=False)
dept_mean_salary.plot.bar(color="#4C72B0")
plt.title("Mean Monthly Salary by Department")
plt.xlabel("Department")
plt.ylabel("Mean Monthly Salary ($)")
plt.xticks(rotation=30)
plt.tight_layout()
plt.savefig(f"{PLOTS}/02_bar_salary_by_dept.png")
plt.close()
print("\nMean salary by department:\n", dept_mean_salary)

# 7c. Histogram of most skewed column
plt.figure(figsize=(8, 5))
sns.histplot(df[most_skewed_col].dropna(), bins=20, kde=True, color="#DD8452")
plt.title(f"Distribution of {most_skewed_col} (most skewed, skew={skew_vals.iloc[0]:.2f})")
plt.xlabel(most_skewed_col)
plt.ylabel("Count")
plt.tight_layout()
plt.savefig(f"{PLOTS}/03_hist_{most_skewed_col}.png")
plt.close()

# 7d. Scatter plot: years_experience vs monthly_salary (expected correlation)
plt.figure(figsize=(8, 5))
sns.scatterplot(data=df, x="years_experience", y="monthly_salary", hue="department", alpha=0.6)
plt.title("Years of Experience vs Monthly Salary")
plt.xlabel("Years of Experience")
plt.ylabel("Monthly Salary ($)")
plt.tight_layout()
plt.savefig(f"{PLOTS}/04_scatter_experience_salary.png")
plt.close()
pearson_exp_sal = df["years_experience"].corr(df["monthly_salary"])
print(f"\nPearson corr(years_experience, monthly_salary) = {pearson_exp_sal:.3f}")

# 7e. Box plot: monthly_salary split by department
plt.figure(figsize=(8, 5))
sns.boxplot(data=df, x="department", y="monthly_salary")
plt.title("Monthly Salary Distribution by Department")
plt.xlabel("Department")
plt.ylabel("Monthly Salary ($)")
plt.xticks(rotation=30)
plt.tight_layout()
plt.savefig(f"{PLOTS}/05_box_salary_by_dept.png")
plt.close()
print("\nMedian salary by department:\n",
      df.groupby("department", observed=True)["monthly_salary"].median().sort_values(ascending=False))

# ---------------------------------------------------------------------------
# Task 7 (cont'd): Correlation heat map
# ---------------------------------------------------------------------------
section("TASK 7 — Correlation heat map (Pearson)")
pearson_corr = df[numeric_cols].corr(method="pearson")
print(pearson_corr.round(3))

plt.figure(figsize=(8, 6.5))
sns.heatmap(pearson_corr, annot=True, fmt=".2f", cmap="coolwarm", center=0)
plt.title("Pearson Correlation Heat Map (Numeric Columns)")
plt.tight_layout()
plt.savefig(f"{PLOTS}/06_heatmap_pearson.png")
plt.close()

# Identify highest |correlation| pair (excluding self-correlation)
corr_pairs = pearson_corr.where(np.triu(np.ones(pearson_corr.shape), k=1).astype(bool)).stack()
top_pair = corr_pairs.abs().sort_values(ascending=False).index[0]
top_pair_val = corr_pairs[top_pair]
print(f"\nHighest |correlation| pair: {top_pair} = {top_pair_val:.3f}")

# ---------------------------------------------------------------------------
# Task 8: Imputation strategy comparison (mean vs median for 2 most-skewed cols)
# ---------------------------------------------------------------------------
section("TASK 8 — Imputation strategy comparison")
top2_skewed = skew_vals.index[:2].tolist()
imputation_report = {}
for col in top2_skewed:
    mean_val = df[col].mean()
    median_val = df[col].median()
    skew_direction = "positive" if skew_vals[col] > 0 else "negative"
    chosen = "median"
    imputation_report[col] = dict(mean=mean_val, median=median_val,
                                   skew=skew_vals[col], direction=skew_direction, chosen=chosen)
    print(f"\n{col}: mean={mean_val:.3f}, median={median_val:.3f}, "
          f"skew={skew_vals[col]:.3f} ({skew_direction}) -> imputing with {chosen}")
    before_nulls = df[col].isnull().sum()
    df[col] = df[col].fillna(median_val)
    print(f"  Nulls before: {before_nulls}, filled with median={median_val:.3f}")

print("\nConfirm no nulls remain in the two columns:")
print(df[top2_skewed].isnull().sum())

# ---------------------------------------------------------------------------
# Task 9 (b): Spearman rank correlation vs Pearson
# ---------------------------------------------------------------------------
section("TASK 9b — Spearman vs Pearson")
spearman_corr = df[numeric_cols].corr(method="spearman")
print("Spearman matrix:\n", spearman_corr.round(3))

diff = (spearman_corr - pearson_corr).abs()
diff_pairs = diff.where(np.triu(np.ones(diff.shape), k=1).astype(bool)).stack()
top3_diff_pairs = diff_pairs.sort_values(ascending=False).head(3)
print("\nTop 3 pairs by |Spearman - Pearson|:")
diff_table_rows = []
for (a, b), d in top3_diff_pairs.items():
    p_val = pearson_corr.loc[a, b]
    s_val = spearman_corr.loc[a, b]
    diff_table_rows.append({"pair": f"{a} vs {b}", "pearson": p_val, "spearman": s_val, "abs_diff": d})
    print(f"  {a} vs {b}: pearson={p_val:.3f}, spearman={s_val:.3f}, |diff|={d:.3f}")
diff_table = pd.DataFrame(diff_table_rows)
print("\n", diff_table)

# ---------------------------------------------------------------------------
# Task 9 (c): Grouped aggregation
# ---------------------------------------------------------------------------
section("TASK 9c — Grouped aggregation")
agg_result = df.groupby("department", observed=True)["monthly_salary"].agg(["mean", "std", "count"])
print(agg_result)

highest_mean_group = agg_result["mean"].idxmax()
highest_std_group = agg_result["std"].idxmax()
mean_ratio = agg_result["mean"].max() / agg_result["mean"].min()
print(f"\nHighest-mean group: {highest_mean_group} (mean={agg_result['mean'].max():.2f})")
print(f"Highest-std group: {highest_std_group} (std={agg_result['std'].max():.2f})")
print(f"Ratio of highest group mean to lowest group mean: {mean_ratio:.3f}")

# ---------------------------------------------------------------------------
# Save cleaned dataset
# ---------------------------------------------------------------------------
section("Saving cleaned_data.csv")
df.to_csv("cleaned_data.csv", index=False)
print("Saved cleaned_data.csv with shape:", df.shape)

# ---------------------------------------------------------------------------
# Dump a machine-readable summary for README generation
# ---------------------------------------------------------------------------
import json
summary = {
    "shape_raw": [1035, 11],
    "shape_after_dedup": list(df.shape),
    "dup_count": int(dup_count),
    "high_null_cols": high_null_cols,
    "null_table": null_table.round(2).to_dict(),
    "mem_before": int(mem_before),
    "mem_after": int(mem_after),
    "skew_vals": skew_vals.round(3).to_dict(),
    "most_skewed_col": most_skewed_col,
    "iqr_report": {k: {kk: float(vv) for kk, vv in v.items()} for k, v in iqr_report.items()},
    "pearson_exp_sal": float(pearson_exp_sal),
    "top_pair": [top_pair[0], top_pair[1], float(top_pair_val)],
    "imputation_report": {k: {kk: (float(vv) if isinstance(vv, (int, float, np.floating)) else vv) for kk, vv in v.items()} for k, v in imputation_report.items()},
    "top3_diff_pairs": diff_table.to_dict(orient="records"),
    "dept_mean_salary": dept_mean_salary.round(2).to_dict(),
    "agg_result": agg_result.round(3).to_dict(),
    "highest_mean_group": highest_mean_group,
    "highest_std_group": highest_std_group,
    "mean_ratio": float(mean_ratio),
}
with open("summary.json", "w") as f:
    json.dump(summary, f, indent=2, default=str)
print("\nsummary.json written for README generation.")
